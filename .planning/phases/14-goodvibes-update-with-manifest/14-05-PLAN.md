    Imports: from __future__ import annotations; hashlib, importlib.metadata, pathlib, shutil; Annotated from typing; typer; Console, Panel from rich; list_template_files and resolve_templates_dir from steps.copy_templates; read_manifest and write_manifest from steps.write_manifest; merge_claude from utils.sentinel_merge; detect_project_type from utils.detect_project_type.

    def update_cmd(
        dry_run: Annotated[bool, typer.Option("--dry-run", help="Preview changes without writing")] = False,
        force: Annotated[bool, typer.Option("--force", help="Skip confirmation and overwrite")] = False,
    ) -> None:
        """Update goodvibes-managed files using the manifest."""

    Flow:
    1. console.rule("[bold]goodvibes update[/bold]")
    2. cwd = pathlib.Path.cwd()
    3. manifest = read_manifest(cwd); if manifest is None: console.print(Panel(UPD-05 message, title="No manifest")), console.rule("Nothing updated."), return (exit 0 — no raise typer.Exit())
    4. template_dir = resolve_templates_dir()
    5. project_type = detect_project_type(cwd)
    6. categorise → (overwrite, skip, net_new) lists (see categorise logic below)
    7. if dry_run: console.print(Panel(3-category summary, title="Dry run — no files written")), console.rule("Run without --dry-run to apply."), return
    8. if not force and overwrite: confirmed = typer.confirm(f"Overwrite {len(overwrite)} managed file(s)?"); if not confirmed: console.rule("Update cancelled."), return
    9. Apply: for rel in overwrite + net_new:
       - Resolve template_src: if rel == 'CLAUDE.md' → template_dir / 'CLAUDE.md'; elif rel == '.github/workflows/ci.yml' → template_dir / '.github' / 'workflows' / f'ci-{project_type}.yml'; else → template_dir / rel
       - if not template_src.exists(): continue (skip — template no longer has this file)
       - if rel == 'CLAUDE.md': template_content = template_src.read_text(encoding="utf-8"); merge_claude(cwd / rel, template_content)
       - else: dest = cwd / rel; dest.parent.mkdir(parents=True, exist_ok=True); shutil.copy2(str(template_src), str(dest))
    10. Write manifest: applied = [r for r in overwrite + net_new if (cwd / r).exists()]; _version = importlib.metadata.version("goodvibes-cli"); write_manifest(cwd, applied, _version)
    11. console.print(Panel("\n".join(overwrite + net_new) or "(none)", title="Updated")), console.rule("[green]Update complete![/green]")

    Categorise logic (inline, no helper function — ponytail):
    - overwrite: list[str] = []
    - skip: list[str] = []
    - net_new: list[str] = []
    - ci_variants = {"ci-node.yml", "ci-python.yml", "ci-both.yml"}
    - selected_variant_src = f"ci-{project_type}.yml"
    - First pass (manifest files → overwrite/skip):
      for rel, manifest_sha in manifest["files"].items():
          dest_path = cwd / rel
          if not dest_path.exists(): overwrite.append(rel); continue
          dest_sha = hashlib.sha256((dest_path).read_bytes()).hexdigest()
          if dest_sha == manifest_sha: overwrite.append(rel)
          else: skip.append(rel)
    - Second pass (template files → net_new):
      all_template_files = list_template_files(template_dir)
      managed_keys = set(manifest["files"].keys())
      for tf in all_template_files:
          if tf == ".goodvibes.json": continue
          # Map CI variant to dest name
          if tf in ci_variants:
              if tf != selected_variant_src: continue  # skip non-selected variants
              dest_rel = ".github/workflows/ci.yml"
          else:
              dest_rel = tf
          if dest_rel not in managed_keys: net_new.append(dest_rel)
